cmake_minimum_required(VERSION 3.20)
project(ax201_hfp_poc LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This diagnostic requires Windows")
endif()
add_executable(ax201_probe app/main.cpp bluetooth/ax201_transport.cpp bluetooth/usb_descriptors.cpp bluetooth/hci_transport.cpp)
target_compile_features(ax201_probe PRIVATE cxx_std_17)
target_compile_definitions(ax201_probe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(ax201_probe PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(ax201_probe PRIVATE setupapi winusb)
if(MSVC)
  target_compile_options(ax201_probe PRIVATE /W4 /WX /utf-8)
endif()
enable_testing()
add_executable(ax201_driver tools/driver_binding.cpp)
target_compile_features(ax201_driver PRIVATE cxx_std_17)
target_compile_definitions(ax201_driver PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(ax201_driver PRIVATE setupapi newdev cfgmgr32)
add_executable(descriptor_tests tests/descriptor_tests.cpp bluetooth/usb_descriptors.cpp)
target_compile_features(descriptor_tests PRIVATE cxx_std_17)
target_include_directories(descriptor_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME usb_descriptor_validation COMMAND descriptor_tests)
add_executable(audio_ring_tests tests/audio_ring_tests.cpp)
target_compile_features(audio_ring_tests PRIVATE cxx_std_17)
target_include_directories(audio_ring_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME audio_ring_clock_drift COMMAND audio_ring_tests)
